"""Render a brand × year DONUT matrix for the audited smartphone research list.

Primary visual encoding (identical geometry law to the pie version):
- x = canonical year (2021-2026), y = smartphone brand
- wedge = one of six primary research directions
- each cell is a DONUT (ring): outer radius r = R_MAX * sqrt(cell_total / max_total),
  so the OUTER circle area is proportional to the total eligible publication count
- the total n= label sits in the hollow centre of every ring
- percentage share of each direction is placed just OUTSIDE the ring, without leaders
- zero-count cells show ``0`` and contain no ring

The script consumes classified.tsv generated by classify_six.py and writes the donut
figure(s) to writing_space/figures plus a note appended to HANDOFF_TO_WRITER.md.
"""
from __future__ import annotations

import json
import math
from pathlib import Path

import matplotlib as mpl
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
from matplotlib.patches import Patch
from matplotlib.patheffects import withStroke

BASE = Path("/home/novix/workspace")
PROJECT = BASE / "project" / "tool_0819_analysis"
FIGURES = BASE / "writing_space" / "figures"
CLASSIFIED = PROJECT / "classified.tsv"
SOURCE_URL = "https://github.com/baolinv0/Tool/blob/main/0819-method.md"
YEARS = list(range(2021, 2027))
BRANDS = ["Huawei", "vivo", "OPPO", "Xiaomi", "HONOR"]
CATEGORIES = [
    "ISP / RAW / HDR / 3A",
    "Restoration & Zoom",
    "Portrait & Depth",
    "Editing & Retouching",
    "Quality & Aesthetics",
    "Capture Intelligence",
]
COLORS = {
    "ISP / RAW / HDR / 3A": "#2a78d6",
    "Restoration & Zoom": "#eb6834",
    "Portrait & Depth": "#1baf7a",
    "Editing & Retouching": "#eda100",
    "Quality & Aesthetics": "#e87ba4",
    "Capture Intelligence": "#008300",
}
INK = "#0b0b0b"
SECONDARY = "#52514e"
MUTED = "#898781"
GRID = "#e1e0d9"
SURFACE = "#fcfcfb"
PAGE = "#f9f9f7"

# Outer radius law: r = R_MAX * sqrt(N / N_max), outer area proportional to N.
R_MAX = 0.66
# Symmetric per-cell data limits applied after pie() so every ring is drawn on the
# same data scale and outside labels never leave the cell.
CELL_LIMIT = 1.00
# Radial gap between the outer ring edge and the outside percentage labels.
LABEL_GAP = 0.10
# Ring-thickness rule in data units: width = clip(0.17*r, MIN, MAX).
# A floor of ~0.055 keeps even n=1 cells visible as a ring with a usable centre;
# capping avoids paper-thin rings on the largest cells.
RING_W_MIN = 0.055
RING_W_MAX = 0.115
RING_W_FRAC = 0.17
PERCENT_DECIMALS = 0

FIGURES.mkdir(parents=True, exist_ok=True)
mpl.rcParams.update({
    "font.family": "DejaVu Sans",
    "font.size": 10,
    "axes.facecolor": SURFACE,
    "figure.facecolor": PAGE,
    "savefig.facecolor": PAGE,
    "axes.edgecolor": "#c3c2b7",
    "axes.linewidth": 0.7,
    "xtick.color": MUTED,
    "ytick.color": MUTED,
    "axes.labelcolor": SECONDARY,
})


def load_matrix() -> pd.DataFrame:
    if not CLASSIFIED.exists():
        raise FileNotFoundError(f"Missing {CLASSIFIED}; run classify_six.py first")
    df = pd.read_csv(CLASSIFIED, sep="\t")
    df["duplicate"] = df["duplicate"].astype(str).str.lower().eq("true")
    df["year"] = pd.to_numeric(df["year"], errors="raise").astype(int)
    eligible = df.loc[(~df["duplicate"]) & df["category"].isin(CATEGORIES)].copy()
    eligible["category"] = pd.Categorical(eligible["category"], categories=CATEGORIES, ordered=True)
    eligible["company"] = pd.Categorical(eligible["company"], categories=BRANDS, ordered=True)
    matrix = (eligible.groupby(["company", "year", "category"], observed=False).size()
              .rename("papers").reset_index())
    matrix["company"] = pd.Categorical(matrix["company"], categories=BRANDS, ordered=True)
    matrix["year"] = pd.Categorical(matrix["year"], categories=YEARS, ordered=True)
    matrix["category"] = pd.Categorical(matrix["category"], categories=CATEGORIES, ordered=True)
    full_index = pd.MultiIndex.from_product([BRANDS, YEARS, CATEGORIES], names=["company", "year", "category"])
    matrix = (matrix.set_index(["company", "year", "category"])["papers"]
              .reindex(full_index, fill_value=0).rename("papers").reset_index())
    matrix["papers"] = matrix["papers"].astype(int)
    totals = (matrix.groupby(["company", "year"], observed=False)["papers"].sum()
              .rename("total_papers").reset_index())
    matrix = matrix.merge(totals, on=["company", "year"], how="left")
    matrix["share"] = np.where(matrix["total_papers"] > 0,
                                matrix["papers"] / matrix["total_papers"], 0.0)
    matrix["n_max"] = int(totals["total_papers"].max())
    matrix["radius"] = np.where(matrix["total_papers"] > 0,
                                 R_MAX * np.sqrt(matrix["total_papers"] / matrix["n_max"]), 0.0)
    return matrix


def ring_width_for(radius: float) -> float:
    """Return the ring thickness (data units) for a given outer radius."""
    if radius <= 0:
        return 0.0
    return float(min(RING_W_MAX, max(RING_W_MIN, RING_W_FRAC * radius)))


def save(fig: mpl.figure.Figure, stem: str) -> None:
    png = FIGURES / f"{stem}.png"
    pdf = FIGURES / f"{stem}.pdf"
    fig.savefig(png, dpi=300, bbox_inches="tight", facecolor=PAGE)
    fig.savefig(pdf, bbox_inches="tight", facecolor=PAGE)
    print(f"generated {png}")
    print(f"generated {pdf}")
    plt.close(fig)


def matrix_donut_figure(matrix: pd.DataFrame) -> None:
    fig, axes = plt.subplots(
        len(BRANDS), len(YEARS), figsize=(22.0, 15.2),
        sharex=True, sharey=True, facecolor=PAGE,
        gridspec_kw={"wspace": 0.15, "hspace": 0.17},
    )
    axes = np.asarray(axes)
    n_max = int(matrix["n_max"].iloc[0])

    for yi, brand in enumerate(BRANDS):
        for xi, year in enumerate(YEARS):
            ax = axes[yi, xi]
            cell = matrix[(matrix["company"].astype(str) == brand) &
                          (matrix["year"].astype(str) == str(year))]
            vals = cell.set_index("category")["papers"].reindex(CATEGORIES, fill_value=0).to_numpy()
            total = int(vals.sum())
            r_outer = 0.0 if total == 0 else R_MAX * math.sqrt(total / n_max)
            r_inner = 0.0 if total == 0 else r_outer - ring_width_for(r_outer)
            ax.set_aspect("equal")
            ax.set_xticks([])
            ax.set_yticks([])
            for spine in ax.spines.values():
                spine.set_visible(False)

            if total == 0:
                ax.text(0, 0, "0", ha="center", va="center", color=MUTED,
                        fontsize=14, weight="semibold")
            else:
                nz = vals > 0
                nz_vals = vals[nz]
                nz_categories = [c for c, keep in zip(CATEGORIES, nz) if keep]
                wedges, _ = ax.pie(
                    nz_vals, radius=r_outer,
                    startangle=90, counterclock=False,
                    colors=[COLORS[c] for c in nz_categories],
                    wedgeprops={"width": r_outer - r_inner,
                                "linewidth": 1.5, "edgecolor": SURFACE},
                )

                # Single-slice ring: place the percentage above the ring.
                single = len(wedges) == 1
                for idx, (wedge, value) in enumerate(zip(wedges, nz_vals)):
                    share = value / total
                    if single:
                        mid_deg = 90.0
                    else:
                        mid_deg = (wedge.theta1 + wedge.theta2) / 2.0
                    mid = math.radians(mid_deg)
                    ux, uy = math.cos(mid), math.sin(mid)
                    label_radius = r_outer + LABEL_GAP
                    ax.text(
                        label_radius * ux, label_radius * uy,
                        f"{share:.{PERCENT_DECIMALS}%}",
                        ha="center", va="center", fontsize=7.8,
                        color=INK, weight="bold", zorder=7,
                        path_effects=[withStroke(linewidth=2.2, foreground=SURFACE)],
                    )

                # n= in the hollow centre. Font scales modestly so short labels in
                # the smallest cells (n<=3) fit inside the inner hole.
                font_size = 9.0 if r_outer >= 0.30 else (7.8 if r_outer >= 0.18 else 7.0)
                ax.text(
                    0, 0, f"n={total}", ha="center", va="center",
                    fontsize=font_size, color=INK, weight="bold", zorder=9,
                    path_effects=[withStroke(linewidth=2.6, foreground=SURFACE)],
                )

            # Symmetric limits so all rings share the same data scale and outside
            # labels stay inside the cell.
            ax.set_xlim(-CELL_LIMIT, CELL_LIMIT)
            ax.set_ylim(-CELL_LIMIT, CELL_LIMIT)
            ax.axhline(-CELL_LIMIT, color=GRID, lw=0.5, zorder=0, clip_on=False)
            ax.axvline(-CELL_LIMIT, color=GRID, lw=0.5, zorder=0, clip_on=False)
            if yi == 0:
                ax.set_title(str(year), color=INK, fontsize=13.5, weight="semibold", pad=11)
            if xi == 0:
                ax.text(-0.10, 0.5, brand, transform=ax.transAxes, ha="right", va="center",
                        fontsize=13.5, color=INK, weight="semibold", clip_on=False)

    handles = [Patch(facecolor=COLORS[c], edgecolor="none", label=c) for c in CATEGORIES]
    fig.legend(handles=handles, loc="lower center", ncol=3, bbox_to_anchor=(0.5, 0.047),
               frameon=False, fontsize=11, labelcolor=SECONDARY,
               handlelength=1.2, columnspacing=1.8)
    fig.suptitle("Brand × year research matrix: publication mix and volume (donut)",
                 x=0.055, y=0.978, ha="left", fontsize=21, color=INK, weight="semibold")
    fig.text(0.055, 0.948,
             "Outer ring area ∝ total eligible publications · each direction labelled with its rounded share · n= is centred in the ring",
             fontsize=11, color=MUTED)
    fig.text(0.055, 0.075,
             f"Radius: r = {R_MAX:.2f} × √(cell total / maximum cell total). Ring thickness varies with cell size. Percentages sit beside the ring without leader lines; empty cells show 0.",
             fontsize=10, color=SECONDARY)
    fig.text(0.055, 0.023,
             "Counts: audited six-category records only; exact-title duplicates excluded. Percentages are rounded independently, so labels may not sum to exactly 100%. 2026 is a source snapshot.",
             fontsize=9.2, color=MUTED)
    fig.subplots_adjust(left=0.12, right=0.985, top=0.90, bottom=0.16)
    save(fig, "cvpr_brand_year_donut_2021_2026")


def donut_size_key(matrix: pd.DataFrame) -> None:
    fig, ax = plt.subplots(figsize=(11, 2.4), facecolor=PAGE)
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.axis("off")
    n_max = int(matrix["n_max"].iloc[0])
    examples = [1, 5, 10, n_max]
    x_positions = np.linspace(0.25, 0.84, len(examples))
    # Draw representative donut rings in a 0-1 coordinate canvas using circles.
    scale = 0.30 / R_MAX  # map matrix data radius to the key canvas
    for x, n in zip(x_positions, examples):
        r_o = R_MAX * math.sqrt(n / n_max)
        r_i = r_o - ring_width_for(r_o)
        ax.add_patch(plt.Circle((x, 0.60), r_o * scale, facecolor="#c9c7c1",
                                edgecolor="#52514e", linewidth=0.8))
        ax.add_patch(plt.Circle((x, 0.60), r_i * scale, facecolor=PAGE,
                                edgecolor="none"))
        ax.text(x, 0.14, f"n={n}\nr={r_o:.3f}", ha="center", va="top",
                fontsize=9.5, color=INK)
    ax.text(0.015, 0.60, "Ring-size key", ha="left", va="center", fontsize=11,
            color=INK, weight="semibold")
    ax.text(0.98, 0.92, "Outer circle area is proportional to total publications",
            ha="right", va="center", fontsize=9.5, color=SECONDARY)
    save(fig, "cvpr_brand_year_donut_size_key")


def update_handoff(matrix: pd.DataFrame) -> None:
    n_max = int(matrix["n_max"].iloc[0])
    nonzero_wedges = int((matrix["papers"] > 0).sum())
    zero_cells = int((matrix[["company", "year", "total_papers"]].drop_duplicates()["total_papers"] == 0).sum())
    cells = len(BRANDS) * len(YEARS)
    path = PROJECT / "HANDOFF_TO_WRITER.md"
    existing = path.read_text(encoding="utf-8") if path.exists() else "# Handoff — smartphone research trend visualizations\n"
    marker = "## Brand × year DONUT matrix (current primary matrix visualization)\n"
    if marker in existing:
        existing = existing.split(marker, 1)[0].rstrip() + "\n\n" + marker
    else:
        existing = existing.rstrip() + "\n\n" + marker
    section = f"""
- `cvpr_brand_year_donut_2021_2026.png/pdf`: x-axis = year (2021–2026), y-axis = brand; each cell is one donut (ring).
- `cvpr_brand_year_donut_size_key.png/pdf`: compact outer-radius / ring-thickness key for the donut matrix.
- Encoding: the ring's OUTER radius uses `r = {R_MAX:.2f} × sqrt(cell_total / maximum_cell_total)` with `maximum_cell_total = {n_max}`; therefore the outer circle area is proportional to the cell's eligible-publication count.
- Ring thickness: `clip(0.17 × r, {RING_W_MIN}, {RING_W_MAX})` data units, so even the smallest cells (n=1) render as a ring with a hollow centre while large cells do not get paper-thin rings.
- Centre label: `n=` is drawn inside the hollow centre of every non-zero ring; font scales modestly so short totals in tiny cells stay readable.
- Percentage labels: all `{nonzero_wedges}` non-zero direction wedges carry integer-percent labels placed just outside the outer edge, on each wedge's radial centre line, with a white halo and NO leader lines.
- Zero handling: `{zero_cells}` of `{cells}` brand×year cells have zero eligible records; these cells contain no ring and display a muted `0`.
- Palette and legend are unchanged (six fixed direction colours, one shared legend).
- Data tables (`brand_year_matrix_counts.csv`, `brand_year_totals.csv`, etc.) are unchanged and describe the same underlying counts.
- This donut version supersedes the earlier solid-pie matrix (`smartphone_research_brand_year_matrix_pies.png/pdf`) as the primary matrix visualisation.

### Donut reproduction

```bash
python3 tool_0819_analysis/classify_six.py
python3 tool_0819_analysis/generate_brand_year_donut_matrix.py
```

The 2026 column is a source snapshot through the audit baseline and should not be treated as a complete-year estimate.
"""
    path.write_text(existing + section, encoding="utf-8")


def main() -> None:
    matrix = load_matrix()
    matrix_donut_figure(matrix)
    donut_size_key(matrix)
    update_handoff(matrix)
    print(json.dumps({
        "figure": "cvpr_brand_year_donut_2021_2026.png/pdf",
        "size_key": "cvpr_brand_year_donut_size_key.png/pdf",
        "maximum_cell_total": int(matrix["n_max"].iloc[0]),
        "ring_width_rule": f"clip(0.17*r, {RING_W_MIN}, {RING_W_MAX})",
    }, ensure_ascii=False))


if __name__ == "__main__":
    main()
